// TEMPERATURE
// ============================================================================

void temp_init() {
  adc_init();
  adc_set_temp_sensor_enabled(true);
  adc_select_input(4);
}

float read_temperature() {
  adc_select_input(4);
  uint16_t adc_raw = adc_read();
  
  const float conversion = 3.3f / 4096.0f;
  float voltage = adc_raw * conversion;
  float temp_c = 27.0f - (voltage - 0.706f) / 0.001721f;
  
  return temp_c;
}
